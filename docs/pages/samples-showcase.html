samples/v7/table

1. Data Table Renderer (Table Simple)

Open v7 Table Demo

Uses window.ks.declarative.definitions.data.table.simple to provide instructions and baseFromSpec: "tableSimple", passing the generated spec into window.ks.jsonToDom.appendToDom. 

const localInstructions = window.ks.declarative.definitions.data.table.simple.instructions;
const localBaseFromSpec = window.ks.declarative.definitions.data.table.simple.baseFromSpec;

const result = window.ks.jsonRenderers.runInstructions.steps.runInstructions({
    instructions: localInstructions,
    baseKey: localBaseFromSpec,
    data: { columns, data },
    showLog: false
});

window.ks.jsonToDom.appendToDom({
    jsonSpec: result,
    targetHtmlId: "dom-render-container",
    appendPosition: "replace",
    showLog: false
});
