Multi-Step Pipeline

1. Data Table Renderer (Table Simple)

Open Table Demo

A 3-step sequential pipeline: buildColgroup (1D) sets up widths, buildHeader (1D) produces columns & headers, and buildBody (2D) computes row × column cells into <tbody>. 

import runInstructions from "json-renderers-run-instructions-steps";
import { spec } from "spec-repo";

const base = structuredClone(spec.tableSimple);
const result = runInstructions({
    instructions: ["buildColgroup", "buildHeader", "buildBody"],
    base,
    data: { columns, data },
    showLog: true
});
[
  "buildColgroup",  // 1D: maps columns -> table.colgroup.children (<col>)
  "buildHeader",    // 1D: maps columns -> table.thead.tr.children (<th>)
  "buildBody"       // 2D: maps rows x columns -> table.tbody.children (<tr><td>)
]
{
  "tagName": "table",
  "attributes": { "class": "table" },
  "children": [
    { "tagName": "colgroup", "children": [ { "tagName": "col" }, { "tagName": "col" } ] },
    { "tagName": "thead", "children": [ { "tagName": "tr", "children": [ { "tagName": "th", "innerText": "ID" } ] } ] },
    { "tagName": "tbody", "children": [ { "tagName": "tr", "children": [ { "tagName": "td", "innerText": "1" } ] } ] }
  ]
}
